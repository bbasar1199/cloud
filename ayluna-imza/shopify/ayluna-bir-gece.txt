(function(){
  var kok=document.getElementById('ay-{{ section.id }}'); if(!kok) return;
  var kap=kok.querySelector('.makaralar'), kartlar=[].slice.call(kok.querySelectorAll('.makara')), cubuk=[].slice.call(kok.querySelectorAll('.ilerleme i'));
  var az=matchMedia('(prefers-reduced-motion: reduce)').matches;
  var io=new IntersectionObserver(function(es){ es.forEach(function(e){ var v=e.target; if(e.isIntersecting){ var p=v.play(); if(p&&p.catch)p.catch(function(){}); } else v.pause(); }); },{threshold:.35});
  function hazirla(){ [].forEach.call(kok.querySelectorAll('video'),function(v){ v.muted=true; if(az){ v.removeAttribute('autoplay'); v.preload='none'; v.pause(); } else io.observe(v); }); }
  function egim(){ var orta=kap.scrollLeft+kap.clientWidth/2, en=0, yakin=1e9;
    kartlar.forEach(function(m,i){ var c=m.offsetLeft+m.offsetWidth/2, o=(c-orta)/m.offsetWidth, oo=Math.max(-1.4,Math.min(1.4,o));
      m.style.transform=az?'':'rotateY('+(-oo*22)+'deg) scale('+(1-Math.abs(oo)*.08)+')'; m.style.opacity=1-Math.min(.45,Math.abs(oo)*.35);
      if(yakin>Math.abs(o)){yakin=Math.abs(o);en=i;} });
    cubuk.forEach(function(x,i){x.style.setProperty('--d',i<=en?1:0);}); }
  kap.addEventListener('scroll',function(){requestAnimationFrame(egim);},{passive:true});
  function model(m){ var k=m==='erkek'?'e':'k';
    kartlar.forEach(function(a){ var src=a.dataset[k+'Src']||a.dataset.kSrc, po=a.dataset[k+'Poster']||a.dataset.kPoster, v=a.querySelector('video');
      a.querySelector('.ay-b').textContent=a.dataset[k+'B']; a.querySelector('.ay-a').textContent=a.dataset[k+'A'];
      if(v&&src&&v.querySelector('source').getAttribute('src')!==src){ v.querySelector('source').setAttribute('src',src); if(po) v.poster=po; v.load(); if(!az){ var r=v.getBoundingClientRect(); if(r.right>0&&innerWidth>r.left&&r.bottom>0&&innerHeight>r.top){ var p=v.play(); if(p&&p.catch)p.catch(function(){}); } } } }); }
  document.addEventListener('ayluna:model',function(e){model(e.detail.model);});
  hazirla(); if(window.AylunaModel&&window.AylunaModel!=='kiz') model(window.AylunaModel); egim();
})();
</script>
